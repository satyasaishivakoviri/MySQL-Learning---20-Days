Triggers

-->A trigger is a special type of stored program that automatically executes in response to specific event INSERT,UPDATE,DELETE occurring on a specified table.unlike stored procedure.A triiger is never called directly .The database enginee invokes it automactaically when the trigerring event happens
Syntax:
DELIMETER //
CREATE TRIGGER trigger_name
{BEFORE|AFTER}{INSERT|UPDATE|DELETE}
on table_name FOR EACH ROW
BEGIN
  -----trigger logic
  -----use new.column-name for inserted/updated values
  ----use old.column-name for deleted/previous values
END //
DELIMETER ;

BEFORE: Trigger fires before the triggering event applied to the table. often used for validation
AFTER: Trigger fires after the triggering event has been applied.Often used for auditing/logging
NEW: Return the new row values. Available in INSERT and UPDATE triggers
OLD: refers to the row values before the change available on UPDATE and DELETE
FOR EACH ROW: Required in MYSQL the trigger body runs once per affected row

CREATE TABLE Employee_Audit(
    audit_id INT AUTO_INCREMENT PRIMARY KEY,
    emp_id INT,
    action_type VARCHAR(20),
    old_salary DECIMAL(10,2),
    new_salary DECIMAL(10,2),
    action_time DATETIME
);


-- Don't allow employees salaray less than 20000

DELIMITER //
CREATE TRIGGER TRG_BEFORE_INSERT_SAL
BEFORE INSERT
ON EMPLOYEES
FOR EACH ROW
BEGIN 
    IF NEW.SALARY<20000 THEN
    SIGNAL SQLSTATE '45000'
    SET MESSAGE_TEXT='SALARY CANNOT BE LESS THAN 20000';
    END IF;
END //

DELIMETER ;
INSERT INTO EMPLOYEES VALUES(106,'VASU','IT',15000,2,'DELHI');

ERROR 1644 (45000): SALARY CANNOT BE LESS THAN 20000

CREATE TRIGGER TRG_BEFORE_update_salary
BEFORE UPDATE 
ON EMPLOYEES
FOR EACH ROW
BEGIN 
     IF NEW.SALARY<OLD.SALARY THEN
SIGNAL SQLSTATE '45000'
SET MESSAGE_TEXT='SALARY CANNOT BE REDUCED';
END IF;
END //
